cmake_minimum_required(VERSION 3.16)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(overlay-cli LANGUAGES CXX)

if(APPLE)
  enable_language(OBJCXX)
  set(PLATFORM_SOURCE platform_mac.mm)
elseif(WIN32)
  set(PLATFORM_SOURCE platform_win.cpp)
else()
  message(FATAL_ERROR "overlay-cli supports only macOS and Windows")
endif()
add_library(overlay-native STATIC common.cpp ${PLATFORM_SOURCE})
target_compile_features(overlay-native PUBLIC cxx_std_17)
target_include_directories(overlay-native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(overlay-native PROPERTIES CXX_EXTENSIONS OFF)
if(APPLE)
  target_compile_options(overlay-native PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  set_target_properties(overlay-native PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
  target_link_libraries(overlay-native PUBLIC "-framework AppKit" "-framework ImageIO")
else()
  target_compile_definitions(overlay-native PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(overlay-native PUBLIC user32 gdi32 windowscodecs ole32)
endif()
add_executable(overlay-cli main.cpp)
target_link_libraries(overlay-cli PRIVATE overlay-native)
if(MSVC)
  target_compile_options(overlay-native PUBLIC /utf-8 /W4)
elseif(MINGW)
  target_link_options(overlay-cli PRIVATE -municode)
else()
  target_compile_options(overlay-native PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(overlay-native-tests tests.cpp)
  target_link_libraries(overlay-native-tests PRIVATE overlay-native)
  add_test(NAME native-validation COMMAND overlay-native-tests)
  set_tests_properties(native-validation PROPERTIES TIMEOUT 20)
  add_test(NAME protocol-version COMMAND overlay-cli --version)
  set_tests_properties(protocol-version PROPERTIES PASS_REGULAR_EXPRESSION "\"version\":\"[0-9]+\\.[0-9]+\\.[0-9]+\"")
  add_test(NAME protocol-capabilities COMMAND overlay-cli capabilities)
  set_tests_properties(protocol-capabilities PROPERTIES PASS_REGULAR_EXPRESSION "\"modal\":false,\"format\":\"png\",\"modes\":\\[\"passive\",\"clickable\"\\]")
endif()
